    <script>
        // ============================================================
        //  CONFIG
        // ============================================================
        const API_URL = 'https://draw.ar-lottery01.com/WinGo/WinGo_1M/GetHistoryIssuePage.json?ts=';
        const WIN_GO_URL = 'https://www.jaiclub48.com/#/saasLottery/WinGo?gameCode=WinGo_30S&lottery=WinGo';
        const TG_URL = 'https://t.me/+T1cr_hi2iilkZDY1';

        // ============================================================
        //  STATE
        // ============================================================
        let showNumber = false;
        let isScanning = false;
        let currentPeriod = '';
        let cachedResult = null; // { period, display, raw }

        // ============================================================
        //  DOM REFS
        // ============================================================
        const miniBtn = document.getElementById('miniBtn');
        const mainPanel = document.getElementById('main');
        const screenEl = document.getElementById('innerScreen');
        const btnArea = document.getElementById('btnArea');
        const scanLine = document.getElementById('scanLine');

        // ============================================================
        //  PERIOD HELPER
        // ============================================================
        function getPeriodId() {
            const d = new Date();
            const y = d.getUTCFullYear();
            const mo = String(d.getUTCMonth() + 1).padStart(2, '0');
            const day = String(d.getUTCDate()).padStart(2, '0');
            const totalMin = d.getUTCHours() * 60 + d.getUTCMinutes();
            return y + mo + day + '1000' + (10001 + totalMin);
        }

        // ============================================================
        //  API — FETCH LATEST RESULT
        // ============================================================
        async function fetchLatestResult() {
            const res = await fetch(API_URL + Date.now(), { cache: 'no-store' });
            const json = await res.json();

            // Try multiple possible response shapes
            let list = json?.data?.list || json?.data || json?.list || json?.rows || json?.data?.records || [];
            if (!Array.isArray(list) || !list.length) throw new Error('No data from API');

            // Sort by issueNumber descending
            list.sort((a, b) => {
                const pa = String(a.issueNumber || a.issue || a.period || a.periodNo || a.Id || '');
                const pb = String(b.issueNumber || b.issue || b.period || b.periodNo || b.Id || '');
                return pb.localeCompare(pa);
            });

            const latest = list[0];
            const period = String(latest.issueNumber || latest.issue || latest.period || latest.periodNo || latest.Id ||
                '');
            const raw = latest.number ?? latest.result ?? latest.openCode ?? latest.open ?? latest.code;
            if (!period || raw === undefined) throw new Error('Invalid data');

            const numVal = parseInt(String(raw), 10);
            const isBig = !isNaN(numVal) && numVal >= 5;
            const label = isBig ? 'BIG' : 'SMALL';
            const display = !isNaN(numVal) ? label + ' ' + numVal : String(raw);

            return { period, display, raw: numVal, label };
        }

        // ============================================================
        //  RENDER
        // ============================================================
        function render(state, payload) {
            // state: 'idle' | 'scanning' | 'result' | 'error'
            const screen = screenEl;
            const btns = btnArea;

            // Clear scan line
            scanLine.style.display = 'none';

            if (state === 'idle') {
                screen.innerHTML = `
                    <div class="status-tag">READY</div>
                    <div class="period-display">ID: ----</div>
                    <div class="res-text" style="font-size:20px;">⏳ TAP SCAN</div>
                    <div class="sub-text">Fetch the latest WinGo result</div>
                    <div class="bottom-info">LIVE DATA</div>
                `;
                btns.innerHTML = `
                    <button class="action-btn" onclick="startScan()">⚡ SCAN RESULT</button>
                    <div class="row">
                        <button class="secondary-btn ${showNumber ? 'active' : ''}" onclick="toggleNumber()">
                            NUMBERS: ${showNumber ? 'ON' : 'OFF'}
                        </button>
                        <button class="secondary-btn" onclick="openWinGo()">🎯 OPEN WIN GO</button>
                        <div class="tg-btn" onclick="openTG()">${tgIcon()}</div>
                    </div>
                `;
                return;
            }

            if (state === 'scanning') {
                scanLine.style.display = 'block';
                screen.innerHTML = `
                    <div class="status-tag" style="color:#00ccff;">SCANNING</div>
                    <div class="period-display">ID: ${currentPeriod.slice(-6) || '···'}</div>
                    <div class="res-text" style="font-size:18px; letter-spacing:2px;">FETCHING...</div>
                    <div class="sub-text">Connecting to server</div>
                    <div class="bottom-info">PLEASE WAIT</div>
                `;
                btns.innerHTML = `
                    <button class="action-btn" disabled style="opacity:0.6;">⏳ SCANNING...</button>
                    <div class="row">
                        <button class="secondary-btn ${showNumber ? 'active' : ''}" onclick="toggleNumber()">
                            NUMBERS: ${showNumber ? 'ON' : 'OFF'}
                        </button>
                        <button class="secondary-btn" onclick="openWinGo()">🎯 OPEN WIN GO</button>
                        <div class="tg-btn" onclick="openTG()">${tgIcon()}</div>
                    </div>
                `;
                return;
            }

            if (state === 'result') {
                const p = payload;
                const displayText = p.display || '—';
                const periodId = p.period || '----';

                // Color the BIG/SMALL part
                let htmlDisplay = displayText;
                if (p.label === 'BIG') {
                    htmlDisplay = displayText.replace('BIG', '<span class="big">BIG</span>');
                } else if (p.label === 'SMALL') {
                    htmlDisplay = displayText.replace('SMALL', '<span class="small">SMALL</span>');
                } else {
                    htmlDisplay = displayText;
                }
                // Highlight number
                if (showNumber && p.raw !== undefined && !isNaN(p.raw)) {
                    htmlDisplay = htmlDisplay.replace(/\d+/, '<span class="num">$&</span>');
                } else if (!showNumber && p.label) {
                    // Show only BIG/SMALL without number
                    htmlDisplay = p.label;
                } else if (!showNumber && p.raw === undefined) {
                    // fallback
                    htmlDisplay = displayText;
                }

                screen.innerHTML = `
                    <div class="status-tag" style="color:#00ff88;">✓ CAPTURED</div>
                    <div class="period-display">ID: ${periodId.slice(-6)}</div>
                    <div class="res-text">${htmlDisplay}</div>
                    <div class="sub-text">Period ended — result verified</div>
                    <div class="bottom-info">● LIVE</div>
                `;
                btns.innerHTML = `
                    <button class="action-btn" onclick="startScan()">🔄 REFRESH</button>
                    <div class="row">
                        <button class="secondary-btn ${showNumber ? 'active' : ''}" onclick="toggleNumber()">
                            NUMBERS: ${showNumber ? 'ON' : 'OFF'}
                        </button>
                        <button class="secondary-btn" onclick="openWinGo()">🎯 OPEN WIN GO</button>
                        <div class="tg-btn" onclick="openTG()">${tgIcon()}</div>
                    </div>
                `;
                return;
            }

            if (state === 'error') {
                const msg = payload || 'Something went wrong';
                screen.innerHTML = `
                    <div class="status-tag" style="color:#ff4444;">⚠ ERROR</div>
                    <div class="period-display">ID: ----</div>
                    <div class="res-text" style="font-size:16px; color:#ff6666;">${msg}</div>
                    <div class="sub-text">Tap RETRY to try again</div>
                    <div class="bottom-info">● OFFLINE</div>
                `;
                btns.innerHTML = `
                    <button class="action-btn" onclick="startScan()">🔄 RETRY</button>
                    <div class="row">
                        <button class="secondary-btn ${showNumber ? 'active' : ''}" onclick="toggleNumber()">
                            NUMBERS: ${showNumber ? 'ON' : 'OFF'}
                        </button>
                        <button class="secondary-btn" onclick="openWinGo()">🎯 OPEN WIN GO</button>
                        <div class="tg-btn" onclick="openTG()">${tgIcon()}</div>
                    </div>
                `;
                return;
            }
        }

        // ============================================================
        //  ACTIONS
        // ============================================================
        function toggleNumber() {
            showNumber = !showNumber;
            // Re-render current state
            if (cachedResult) {
                render('result', cachedResult);
            } else {
                render('idle');
            }
        }

        function openWinGo() {
            if (window.TeamBridge) {
                TeamBridge.loadUrl(WIN_GO_URL);
            } else {
                window.open(WIN_GO_URL, '_blank');
            }
        }

        function openTG() {
            if (window.TeamBridge) {
                TeamBridge.openExternal(TG_URL);
            } else {
                window.open(TG_URL, '_blank');
            }
        }

        function tgIcon() {
            return `<svg viewBox="0 0 24 24"><path d="M12 0C5.373 0 0 5.373 0 12s5.373 12 12 12 12-5.373 12-12S18.627 0 12 0zm5.894 8.221l-1.97 9.28c-.145.658-.537.818-1.084.508l-3-2.21-1.446 1.394c-.16.16-.295.295-.605.295l.213-3.054 5.56-5.022c.242-.213-.054-.333-.373-.121l-6.869 4.326-2.96-.924c-.643-.203-.658-.643.136-.953l11.55-4.458c.535-.199 1.005.128.844.943z"/></svg>`;
        }

        // ============================================================
        //  SCAN
        // ============================================================
        async function startScan() {
            if (isScanning) return;
            isScanning = true;
            currentPeriod = getPeriodId();

            render('scanning');

            try {
                const data = await fetchLatestResult();
                cachedResult = data;
                currentPeriod = data.period || currentPeriod;
                render('result', data);
            } catch (err) {
                console.warn('Scan error:', err);
                render('error', err.message || 'API fetch failed');
            } finally {
                isScanning = false;
            }
        }

        // ============================================================
        //  PANEL OPEN / CLOSE
        // ============================================================
        function openM() {
            mainPanel.style.display = 'flex';
            miniBtn.style.display = 'none';
            if (window.TeamBridge) TeamBridge.resizeView(260, 360);
            // Reset to idle
            cachedResult = null;
            render('idle');
        }

        function closeM() {
            mainPanel.style.display = 'none';
            miniBtn.style.display = 'flex';
            scanLine.style.display = 'none';
            if (window.TeamBridge) TeamBridge.resizeView(80, 80);
        }

        // ============================================================
        //  AUTO-PERIOD UPDATE (for display)
        // ============================================================
        setInterval(() => {
            const newPeriod = getPeriodId();
            if (newPeriod !== currentPeriod && !isScanning) {
                currentPeriod = newPeriod;
                // If we have a cached result, keep showing it; otherwise idle
                if (!cachedResult) {
                    render('idle');
                }
            }
        }, 3000);

        // ============================================================
        //  INIT
        // ============================================================
        window.onload = function() {
            render('idle');
        };

        // Handle TeamBridge resize on open/close
        if (window.TeamBridge) {
            // initial resize for mini state
            TeamBridge.resizeView(80, 80);
        }
    </script>
